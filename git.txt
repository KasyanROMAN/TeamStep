1. git init - створити репозиторій
2. git clone назва репозиторію - скачати проект з GitHub
3. git add . - зафіксувати зміни
4. git commit -m 'Hello' - зробити коміт
5. git push - запушити до GitHub
6. git push --set-upstream назва_гілки - додати гілку до репозиторію
7. git s - перевірити статус фіксації add
8. git g - перевірити коміти, дату створення і гілки
9. git co -b vetka - створити гілку і одразу бути в ній
10. git br vetka - створити гілку
11. git br - перевірити гілки які існують
12. git br -v - перевірити на якій гілки та останній коміт
13. git br -m назва_гілки - змінити назву гілки
14. git switch vetka або git co vetka - переключити гілку
15. git co . або git co назва_файла - отмєна до НЕ фіксації
16. git reset название_файла або git reset . - отмєна до add
17. git br -D назва - видалити гілку 
18. git co код_коміта - перейти до коміту
19. git log -5 - вивести останні 5 комітів
20. git reset --hard HEAD^1 - відмінити коміт до останнього 
21. git reset --soft HEAD^1 - відмінити коміт але змінити залишити 
22. git pull origin назва_гілки - відтягнути зміни
23. git commit --amend -m "назва коміту" - змінити коміт содєржиме останього коміту
24. git commit --amend --no-edit - залишити старе повідомлення комітузалишити старе повідомлення коміту
25.git merge назва_гілки - слити гілки
26. git log назва_гілки --oneline - подивитися комінити гілки

27. git cherry-pick id - взяти коміт з чужої гілки
28. git stash	Зберегти всі незакомічені зміни
29. git stash save "опис"	Зберегти з описом (старий синтаксис)
30. git stash push -m "опис"	Новий синтаксис з описом
31. git stash list	Показати список збережених stash
32. git stash apply	Повернути останній stash (НЕ видаляє його зі списку)
33. git stash pop	Повернути останній stash і видалити його зі списку
34. git stash apply stash@{1}	Повернути конкретний stash
35. git stash drop stash@{0}	Видалити stash зі списку
36 git stash clear	Видалити всі stash

SHOW

37. git show	Показує останній коміт з diff
38. git show a7e24f5	Показує конкретний коміт
39. git show --stat	Коротко про змінені файли
40. git show HEAD:file.py	Вміст файлу з останнього коміту
41. git rebase назва_гілки - лінійно солити гілку
42. git rebase main	Переносить твої коміти поверх main
43. git pull --rebase	Синхронізація з віддаленим без merge-коміта
44. git rebase -i HEAD~3	Інтерактивне редагування останніх 3 комітів
45. git rebase --abort	Відміна ребейсу, якщо щось пішло не так
46. git rebase --continue	Продовжити ребейс після конфлікту
47. git rebase --skip	Пропустити проблемний коміт
48. git bisect start	Почати сесію пошуку
49. git bisect bad	Позначити поточний коміт як "поганий"
50. git bisect good [хеш/мітка]	Позначити коміт як "хороший"
51. git bisect reset	Завершити сесію bisect і повернутися назад
52. git bisect skip	Пропустити поточний коміт (наприклад, якщо не вдається перевірити)
53. git bisect visualize	Відкрити коміти між good і bad в gitk (GUI)
54. git bisect log	Показати журнал команд, які ти робив під час сесії
55. git bisect replay <файл>	Повторити пошук з лог-файлу bisect
56. git bisect run <команда>	Автоматично перевіряти кожен коміт скриптом або тестом
57. git merge --abort - відміна слиття
58. git cherry-pick --abort - відмінна слиття комітту
59. git restore . - Відкотити всі незбережені зміни до злиття
60. git log --oneline --merges - Показує всі коміти злиття:
61. git merge --abort - Якщо є конфлікти і ти хочеш все відмінити
62. git reset --hard HEAD~1 і git push origin <branch> --force - відміна пуша
63. git revert -m 1 <merge_commit_hash> - відміна злиття після того як зробили коміт або душ
